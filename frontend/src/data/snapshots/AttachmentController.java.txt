package edu.handong.web11.controller;

import edu.handong.web11.domain.Attachment;
import edu.handong.web11.repository.AttachmentRepository;
import edu.handong.web11.service.FileStorageService;
import org.springframework.core.io.Resource;
import org.springframework.http.*;
import org.springframework.web.bind.annotation.*;
import org.springframework.web.server.ResponseStatusException;
import java.nio.charset.StandardCharsets;

@RestController
@RequestMapping("/attachments")
public class AttachmentController {
    private final AttachmentRepository attachments;
    private final FileStorageService storage;
    public AttachmentController(AttachmentRepository attachments, FileStorageService storage) {
        this.attachments = attachments; this.storage = storage;
    }
    // 클래스의 /attachments 경로와 결합하여 GET /attachments/{id}/download 요청을 처리한다.
    @GetMapping("/{id}/download")
    // URL의 첨부파일 ID를 받고, 상태 코드·헤더·파일 본문을 담은 HTTP 응답을 반환한다.
    public ResponseEntity<Resource> download(@PathVariable Long id) {
        // DB에서 첨부파일 메타데이터를 조회한다. ID가 없으면 아래 예외로 HTTP 404를 반환한다.
        Attachment file = attachments.findById(id).orElseThrow(() ->
            new ResponseStatusException(HttpStatus.NOT_FOUND, "첨부파일이 없습니다."));
        // 저장 이름으로 디스크 경로를 찾고 검사하여 PathResource로 받는다. 파일 전체를 여기서 읽지는 않는다.
        Resource resource = storage.load(file);
        // 브라우저가 첨부파일 다운로드로 처리하도록 Content-Disposition 헤더를 구성한다.
        String disposition = ContentDisposition.attachment()
            // 서버의 UUID 저장 이름 대신 원본 이름을 제공하고, 한글 등은 UTF-8로 인코딩한다.
            .filename(file.getOriginalFilename(), StandardCharsets.UTF_8)
            // 완성된 다운로드 정보를 실제 HTTP 헤더에 넣을 문자열로 변환한다.
            .build().toString();
        // 파일을 전송할 HTTP 200 OK 응답을 구성한다.
        return ResponseEntity.ok()
            // 파일 종류에 관계없이 일반 바이너리 데이터로 전송한다.
            .contentType(MediaType.APPLICATION_OCTET_STREAM)
            // 다운로드 처리 방식과 원본 파일명을 응답 헤더로 브라우저에 전달한다.
            .header(HttpHeaders.CONTENT_DISPOSITION, disposition)
            // Resource를 본문으로 지정하면 Spring이 파일 내용을 읽어 HTTP 응답으로 전송한다.
            .body(resource);
    }
    @GetMapping("/{id}/view")
    public ResponseEntity<Resource> view(@PathVariable Long id) {
        // 다운로드와 동일하게 DB에서 메타데이터를 찾고, 없는 ID는 404로 처리한다.
        Attachment file = attachments.findById(id).orElseThrow(() ->
            new ResponseStatusException(HttpStatus.NOT_FOUND, "첨부파일이 없습니다."));
        // 업로드 시 jpg와 jpeg는 모두 image/jpeg로 저장된다. 지원하는 이미지만 화면에 표시한다.
        String contentType = file.getContentType();
        if (!"image/jpeg".equals(contentType) && !"image/png".equals(contentType)
            && !"image/gif".equals(contentType)) {
            throw new ResponseStatusException(HttpStatus.UNSUPPORTED_MEDIA_TYPE, "미리보기를 지원하지 않는 파일입니다.");
        }
        // 기존 Service의 경로 검사와 파일 존재 검사를 재사용한다. 파일이 없으면 404가 발생한다.
        Resource resource = storage.load(file);
        // 이미지 MIME 타입과 inline 헤더로 응답하여 브라우저의 img 요소에서 표시할 수 있게 한다.
        return ResponseEntity.ok()
            .contentType(MediaType.parseMediaType(contentType))
            .header(HttpHeaders.CONTENT_DISPOSITION, ContentDisposition.inline()
                .filename(file.getOriginalFilename(), StandardCharsets.UTF_8).build().toString())
            .body(resource);
    }
}
